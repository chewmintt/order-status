function search() { var input = document.getElementById('accountInput'); var button = document.getElementById('searchButton'); var account = input.value.trim(); clearMessage(); if (!account) { showMessage('♡ กรุณาใส่ @account ก่อนนะคะ ♡'); return; } button.disabled = true; button.textContent = 'loading... ♡'; var url = 'https://script.google.com/macros/s/AKfycbwNzfYqzFVPMbMTw3biZRFy2s-2niSxTe1-tdMpdsGzZYOrhVUz7EB_DjvSBYNxiCOKRw/exec' + '?account=' + encodeURIComponent(account); fetch(url) .then(function(response) { if (!response.ok) { throw new Error('HTTP ' + response.status); } return response.json(); }) .then(function(data) { renderResults(data); }) .catch(function(error) { console.error(error); showMessage( 'ไม่สามารถเชื่อมต่อระบบได้ กรุณาลองใหม่อีกครั้งค่ะ ♡' ); }) .finally(function() { button.disabled = false; button.textContent = 'search ♡'; }); }
